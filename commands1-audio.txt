import wave, numpy as np
w=wave.open("artifacts/audio-working-copy/dj_cara_after_hours.REAL.wav","rb")
n=w.getnframes(); ch=w.getnchannels(); sw=w.getsampwidth()
print(f"frames={n} channels={ch} sampwidth={sw}")
raw=w.readframes(n); w.close()
a=np.frombuffer(raw, dtype=np.int16).reshape(-1, ch)
for c in range(ch):
    lsb=(a[:,c] & 1).astype(np.uint8)
    for order,label in [(lsb,'MSB-first'),(lsb,'LSB-first')]:
        bits=lsb[:8*20000]
        if label=='LSB-first': bits=bits[::1]
        nbytes=len(bits)//8
        b=np.packbits(bits[:nbytes*8]) if label=='MSB-first' else np.packbits(bits[:nbytes*8][::-1])[::-1]
        txt=bytes(b.tolist())
        printable=sum(32<=x<127 for x in txt[:2000])/max(1,len(txt[:2000]))
        print(f"\n--- channel {c} LSB ({label}) — printable ratio first 2000B: {printable:.2%} ---")
        print(txt[:300])
