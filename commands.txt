## Record the analysis environment

{
date --utc --iso-8601=seconds
grep '^PRETTY_NAME=' /etc/os-release
uname -a
locale
sha256sum --version | head -n 1
stat --version | head -n 1
file --version | head -n 1
strings --version | head -n 1
john --list=build-info | head -n 4
unzip -v | head -n 1
python3 --version
} | tee results/environment-and-tools.txt


## Mirroring and Evidence Preservation of Remote Web Resource

date --utc --iso-8601=seconds
wget --no-check-certificate -p -k -P artifacts/fenix-site https://fenix.dpss.inesc-id.pt/index.html



## Record the original artifact’s path, size, timestamps, type, and SHA-256 digest:

find artifacts/fenix-site -type f -exec sh -c '
  for f; do
    stat -c "%n|%s|%y|%z" "$f"
    file "$f"
    sha256sum "$f"
    echo "---"
  done
' sh {} +




## Create copies of artifacts, preserving timestamps
cp -r --preserve=timestamps artifacts/fenix-site artifacts/fenix-site-working-copy

# Verify hashes of the copies
diff <(find artifacts/fenix-site -type f -exec sha256sum {} \; | sed 's|fenix-site|X|' | sort) \
     <(find artifacts/fenix-site-working-copy -type f -exec sha256sum {} \; | sed 's|fenix-site-working-copy|X|' | sort)
